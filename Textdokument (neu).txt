  <script>
    // ====================== CUSTOMIZE HERE ======================
    const GODDESS_NAME = "YOURNAME";          // ← Change this
    const DEFAULT_PREFIX = "Alice's";         // ← Change this
    const DEFAULT_EMOJI = "🐾";               // ← Change this
    // ===========================================================

    document.getElementById("goddess-name").textContent = GODDESS_NAME;

    const roles = ["Puppy", "Simp", "Sissy", "Slave", "Wallet", "Good Boy", "Kitten", "Drained Pet"];
    const numbers = Array.from({length: 200}, (_, i) => String(i).padStart(3, "0"));

    function updatePreview() {
      document.getElementById("preview-name").textContent = document.getElementById("display-name").value;
      document.getElementById("preview-handle").textContent = document.getElementById("username").value;
    }

    document.getElementById("display-name").addEventListener("input", updatePreview);
    document.getElementById("username").addEventListener("input", updatePreview);
    
    document.getElementById("role-select").addEventListener("change", function() {
      const role = this.value;
      document.getElementById("display-name").value = `${DEFAULT_PREFIX} ${role} ${DEFAULT_EMOJI} #${randomNumber()}`;
      document.getElementById("username").value = `@${GODDESS_NAME.toLowerCase()}slave`;
      updatePreview();
    });

    function randomNumber() {
      return numbers[Math.floor(Math.random() * numbers.length)];
    }

    function randomProfile() {
      const role = roles[Math.floor(Math.random() * roles.length)];
      const num = randomNumber();
      document.getElementById("role-select").value = role;
      document.getElementById("display-name").value = `${DEFAULT_PREFIX} ${role} ${DEFAULT_EMOJI} #${num}`;
      document.getElementById("username").value = `@${GODDESS_NAME.toLowerCase()}${role.toLowerCase()}${num}`;
      updatePreview();
    }

    function acceptIdentity() {
      const name = document.getElementById("display-name").value;
      alert(`Identity accepted!\n\nYou are now:\n${name}\n\nYour old self is gone.\n\nNow show your devotion...`);
      
      // Show the tribute button after accepting
      document.getElementById("tributeBtn").style.display = "block";
    }

    function openTribute() {
      document.getElementById("tributeModal").style.display = "flex";
    }

    function closeTribute() {
      document.getElementById("tributeModal").style.display = "none";
    }

    // Close modal when clicking outside
    document.getElementById("tributeModal").addEventListener("click", function(e) {
      if (e.target === this) closeTribute();
    });

    // Generate one random profile on load
    randomProfile();
  </script>
